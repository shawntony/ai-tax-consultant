### 3.1 조정대상지역 판정

**정의**: 주택 가격 상승률이 높거나 청약 경쟁률이 높은 지역

**지정 기관**: 국토교통부

#### 3.1.2 조정대상지역 판정 로직
```javascript
/**
 * 조정대상지역 판정
 * @param {string} address - 주소 (시/군/구)
 * @returns {boolean} 조정대상지역 여부
 */
function isRegulatedArea(address) {
  // 서울 전역 체크
  if (address.includes('서울특별시')) {
    return true;
  }

  // 기타 조정대상지역 체크
  for (const [city, districts] of Object.entries(REGULATED_AREAS_2024)) {
    if (city === '서울특별시') continue; // 이미 체크함

    for (const district of districts) {
      if (address.includes(district)) {
        return true;
      }
    }
  }

  return false;
}
```

**주의사항**:
- 조정대상지역은 수시로 변경됨
- 양도일 기준으로 판정
- 정확한 판정을 위해 국토교통부 공고 확인 필요

### 3.2 주택 수 판정

#### 3.2.1 주택 수 계산
**판정 기준**:
- 1세대가 보유한 국내 주택 수
- 양도 시점 기준 (양도 후가 아님)

**포함 대상**:
- ✅ 아파트, 단독주택, 다가구주택, 연립주택, 다세대주택
- ✅ 주택 분양권 (조정대상지역 내)
- ✅ 주택 입주권 (조정대상지역 내)

**제외 대상**:
- ❌ 상가, 오피스텔 (주거용 아님)
- ❌ 해외 소재 주택
- ❌ 상속 주택 (일정 요건 충족 시)

```javascript
/**
 * 주택 수 계산
 * @param {Array} houses - 보유 주택 목록
 * @returns {number} 주택 수
 */
function countHouses(houses) {
  let count = 0;

  for (const house of houses) {
    // 주택 유형 확인
    const validTypes = ['아파트', '단독주택', '다가구', '연립', '다세대'];
    if (validTypes.includes(house.type)) {
      count++;
    }

    // 분양권/입주권 (조정대상지역만)
    if (['분양권', '입주권'].includes(house.type)) {
      if (house.isRegulated) {
        count++;
      }
    }
  }

  return count;
}
```

**현행법**: 양도소득세 최대 세율 **75%** (지방소득세 제외)

## 5. 중과세율 계산 로직

### 5.1 중과세율 적용 여부 판단
```javascript
/**
 * 중과세율 적용 여부 판단
 * @param {Object} params
 * @returns {Object} 중과세 적용 결과
 */
function shouldApplySurcharge(params) {
  const {
    houseCount,
    isRegulatedArea,
    transferDate
  } = params;

  // 1주택 → 중과세 없음
  if (houseCount === 1) {
    return {
      apply: false,
      reason: '1주택자는 중과세 적용 안 됨'
    };
  }

  // 비조정대상지역 → 중과세 없음
  if (!isRegulatedArea) {
    return {
      apply: false,
      reason: '비조정대상지역은 중과세 적용 안 됨'
    };
  }

  // 2021년 6월 1일 이전 양도 → 중과세 없음
  if (new Date(transferDate) < new Date('2021-06-01')) {
    return {
      apply: false,
      reason: '2021년 6월 1일 이전 양도분은 중과세 적용 안 됨'
    };
  }

  // 중과세 적용
  return {
    apply: true,
    reason: `${houseCount}주택 조정대상지역 중과세 적용`
  };
}
```

### 5.2 중과세율 계산
```javascript
/**
 * 다주택자 중과세율 계산
 * @param {Object} params
 * @returns {Object} 중과세 계산 결과
 */
function calculateSurcharge(params) {
  const {
    taxBase,              // 과세표준
    basicRate,            // 기본세율
    basicTax,             // 기본세액
    houseCount,
    isRegulatedArea,
    transferDate
  } = params;

  // 중과세 적용 여부 판단
  const surchargeCheck = shouldApplySurcharge({ houseCount, isRegulatedArea, transferDate });

  if (!surchargeCheck.apply) {
    return {
      surchargeApplied: false,
      surchargeRate: 0,
      totalRate: basicRate,
      tax: basicTax,
      reason: surchargeCheck.reason
    };
  }

  // 중과세율 결정
  let surchargeRate = 0;
  if (houseCount === 2) {
    surchargeRate = 0.20; // 20%p 추가
  } else if (houseCount >= 3) {
    surchargeRate = 0.30; // 30%p 추가
  }

  // 총 세율 (최대 75% 제한)
  let totalRate = basicRate + surchargeRate;
  totalRate = Math.min(totalRate, 0.75);

  // 중과세액 재계산
  const progressiveDeduction = getProgressiveDeduction(totalRate, taxBase);
  const surchargeTax = Math.floor(taxBase * totalRate) - progressiveDeduction;

  return {
    surchargeApplied: true,
    surchargeRate,
    totalRate,
    tax: surchargeTax,
    basicRate,
    basicTax,
    additionalTax: surchargeTax - basicTax,
    reason: `${houseCount}주택 조정대상지역 (+${(surchargeRate * 100).toFixed(0)}%p)`
  };
}
```

### 5.3 누진공제 재계산 (중과세 시)
```javascript
/**
 * 중과세율에 대한 누진공제 계산
 * @param {number} totalRate - 총 세율 (기본 + 중과)
 * @param {number} taxBase - 과세표준
 * @returns {number} 누진공제액
 */
function getProgressiveDeduction(totalRate, taxBase) {
  // 중과세율 적용 시 누진공제도 조정
  // 간소화: 기본 누진공제 × (총세율 / 기본세율) 비율로 조정

  // 과세표준에 해당하는 기본 누진공제 찾기
  const DEDUCTION_TABLE = [
    { max: 14000000, deduction: 0 },
    { max: 50000000, deduction: 1260000 },
    { max: 88000000, deduction: 5760000 },
    { max: 150000000, deduction: 15440000 },
    { max: 300000000, deduction: 19940000 },
    { max: 500000000, deduction: 25940000 },
    { max: 1000000000, deduction: 35940000 },
    { max: Infinity, deduction: 65940000 }
  ];

  let basicDeduction = 0;
  for (const { max, deduction } of DEDUCTION_TABLE) {
    if (taxBase <= max) {
      basicDeduction = deduction;
      break;
    }
  }

  // 중과세율 적용 시 누진공제 조정 (단순화)
  // 실제로는 복잡한 계산이 필요하지만, 근사치로 처리
  const adjustmentRatio = totalRate / 0.45; // 0.45는 기본 최대세율
  const adjustedDeduction = Math.floor(basicDeduction * adjustmentRatio);

  return adjustedDeduction;
}
```

## 7. 중과세 배제 특례

### 7.1 일시적 2주택 특례
**요건**:
- 기존 주택을 보유한 상태에서 신규 주택 취득
- 신규 주택 취득 후 **3년 이내** 기존 주택 양도
- 양도하는 주택이 조정대상지역 내

**효과**: 2주택이지만 1주택으로 간주 (중과세 배제)

```javascript
function checkTemp2HouseException(params) {
  const {
    existingHousePurchaseDate,
    newHousePurchaseDate,
    transferDate
  } = params;

  // 신규 주택 취득일로부터 3년 이내인지
  const diffInDays = (transferDate - newHousePurchaseDate) / (1000 * 60 * 60 * 24);
  const within3Years = diffInDays <= 1095; // 3년 = 1095일

  return {
    isException: within3Years,
    reason: within3Years ?
      '일시적 2주택 특례 (3년 이내 양도)' :
      '일시적 2주택 기간 초과 (3년 경과)'
  };
}
```

### 7.2 상속 주택 특례
**요건**:
- 상속으로 취득한 주택
- 상속 개시일로부터 **5년 이내** 양도
- 상속 주택이 조정대상지역 내

**효과**: 상속 주택은 주택 수에서 제외
