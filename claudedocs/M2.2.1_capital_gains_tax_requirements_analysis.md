## 3. 세부 계산 로직

### 3.1 보유기간 계산

```javascript
/**
 * 보유기간 계산 (년 단위)
 * @param {Date} 취득일
 * @param {Date} 양도일
 * @returns {number} 보유년수 (소수점 포함)
 */
function calculateHoldingPeriod(acquisitionDate, transferDate) {
  const diffInDays = (transferDate - acquisitionDate) / (1000 * 60 * 60 * 24)
  const holdingYears = diffInDays / 365

  // 검증
  if (diffInDays < 0) {
    throw new Error('양도일이 취득일보다 빠를 수 없습니다')
  }

  return Math.floor(holdingYears) // 년 단위로 내림
}
```

### 3.2 1세대1주택 비과세 판단

```javascript
/**
 * 1세대1주택 비과세 요건 검토
 * @param {Object} params
 * @returns {Object} { exempt: boolean, exemptAmount: number, taxableGain: number }
 */
function check1House1HouseholdExemption(params) {
  const {
    holdingYears,
    residenceYears,
    transferPrice,
    capitalGain,
    houseCount
  } = params

  // 주택수가 1이 아니면 비과세 불가
  if (houseCount !== 1) {
    return {
      exempt: false,
      exemptAmount: 0,
      taxableGain: capitalGain,
      reason: '1세대1주택이 아님'
    }
  }

  // 보유기간 2년 미만
  if (holdingYears < 2) {
    return {
      exempt: false,
      exemptAmount: 0,
      taxableGain: capitalGain,
      reason: '보유기간 2년 미만'
    }
  }

  // 양도가액 12억 이하 → 전액 비과세
  if (transferPrice <= 1200000000) {
    return {
      exempt: true,
      exemptAmount: capitalGain,
      taxableGain: 0,
      reason: '1세대1주택 비과세 (12억 이하)'
    }
  }

  // 양도가액 12억 초과 → 12억 초과분만 과세
  const exemptRatio = 1200000000 / transferPrice
  const exemptAmount = Math.floor(capitalGain * exemptRatio)
  const taxableGain = capitalGain - exemptAmount

  return {
    exempt: false, // 부분 비과세
    exemptAmount,
    taxableGain,
    reason: '1세대1주택 비과세 (12억 초과분 과세)'
  }
}
```

### 3.3 장기보유특별공제 계산

```javascript
/**
 * 장기보유특별공제 계산
 * @param {Object} params
 * @returns {Object} { deductionRate: number, deductionAmount: number }
 */
function calculateLongTermHoldingDeduction(params) {
  const { holdingYears, residenceYears = 0, houseCount, capitalGain } = params

  // 총 보유기간 = 보유년수 + 거주년수 (1세대1주택인 경우)
  const totalYears = holdingYears + (houseCount === 1 ? residenceYears : 0)

  // 3년 미만 → 공제 불가
  if (totalYears < 3) {
    return {
      deductionRate: 0,
      deductionAmount: 0,
      reason: '보유기간 3년 미만'
    }
  }

  // 공제율 테이블
  const DEDUCTION_RATES = {
    oneHouse: [
      { years: 3, rate: 0.12 },
      { years: 4, rate: 0.16 },
      { years: 5, rate: 0.20 },
      { years: 6, rate: 0.24 },
      { years: 7, rate: 0.28 },
      { years: 8, rate: 0.32 },
      { years: 9, rate: 0.36 },
      { years: 10, rate: 0.40 },
      { years: 11, rate: 0.48 },
      { years: 12, rate: 0.56 },
      { years: 13, rate: 0.64 },
      { years: 14, rate: 0.72 },
      { years: 15, rate: 0.80 }
    ],
    multiHouse: [
      { years: 3, rate: 0.06 },
      { years: 4, rate: 0.08 },
      { years: 5, rate: 0.10 },
      { years: 6, rate: 0.12 },
      { years: 7, rate: 0.14 },
      { years: 8, rate: 0.16 },
      { years: 9, rate: 0.18 },
      { years: 10, rate: 0.20 },
      { years: 11, rate: 0.24 },
      { years: 12, rate: 0.28 },
      { years: 13, rate: 0.32 },
      { years: 14, rate: 0.36 },
      { years: 15, rate: 0.40 }
    ]
  }

  // 공제율 선택
  const rates = houseCount === 1 ? DEDUCTION_RATES.oneHouse : DEDUCTION_RATES.multiHouse

  // 해당 년수의 공제율 찾기
  let applicableRate = 0
  for (let i = rates.length - 1; i >= 0; i--) {
    if (totalYears >= rates[i].years) {
      applicableRate = rates[i].rate
      break
    }
  }

  const deductionAmount = Math.floor(capitalGain * applicableRate)

  return {
    deductionRate: applicableRate,
    deductionAmount,
    totalYears,
    reason: `${houseCount === 1 ? '1세대1주택' : '일반'} ${totalYears}년 보유`
  }
}
```

### 3.4 다주택자 중과세율 적용

```javascript
/**
 * 다주택자 중과세율 계산
 * @param {Object} params
 * @returns {Object} { surchargeApplied: boolean, totalRate: number, tax: number }
 */
function applySurchargeForMultipleHouses(params) {
  const {
    taxBase,
    basicRate,
    houseCount,
    isRegulatedArea,
    basicTax
  } = params

  // 1주택자 또는 비조정대상지역 → 중과세 없음
  if (houseCount === 1 || !isRegulatedArea) {
    return {
      surchargeApplied: false,
      surchargeRate: 0,
      totalRate: basicRate,
      tax: basicTax
    }
  }

  // 중과세율 결정
  let surchargeRate = 0
  if (houseCount === 2) {
    surchargeRate = 0.20 // 20%p 추가
  } else if (houseCount >= 3) {
    surchargeRate = 0.30 // 30%p 추가
  }

  // 총 세율 = 기본세율 + 중과세율
  const totalRate = Math.min(basicRate + surchargeRate, 0.75) // 최대 75%

  // 중과세액 재계산
  const surchargeTax = Math.floor(taxBase * totalRate) - getProgressiveDeduction(totalRate, taxBase)

  return {
    surchargeApplied: true,
    surchargeRate,
    totalRate,
    tax: surchargeTax,
    reason: `${houseCount}주택 조정대상지역`
  }
}
```

## 4. 엣지 케이스 (Edge Cases)

### 4.1 양도손실 케이스
**시나리오**: 취득가액보다 낮은 가격에 양도
**예시**:
- 취득가액: 5억원
- 양도가액: 4억원
- 양도차익: -1억원 (손실)

**처리**:
- 양도소득세 = 0원
- 손실은 다른 양도소득과 상계 가능 (향후 구현)

### 4.2 매우 짧은 보유기간
**시나리오**: 보유기간 1년 미만
**예시**:
- 취득일: 2024-01-01
- 양도일: 2024-06-01
- 보유기간: 5개월

**처리**:
- 1세대1주택 비과세 불가 (2년 미만)
- 장기보유특별공제 불가 (3년 미만)
- 기본세율만 적용

### 4.3 고가주택 (12억 초과)
**시나리오**: 1세대1주택이지만 15억에 양도
**예시**:
- 주택수: 1
- 보유기간: 5년
- 양도가액: 15억원
- 취득가액: 10억원
- 양도차익: 5억원

**처리**:
- 12억 이하분 비과세 비율 = 12억 / 15억 = 80%
- 비과세 양도차익 = 5억 × 80% = 4억원
- 과세 대상 = 5억 - 4억 = 1억원
- 1억원에 대해서만 과세

### 4.4 다주택자 조정대상지역
**시나리오**: 3주택 보유자가 조정대상지역 주택 양도
**예시**:
- 주택수: 3
- 조정대상지역: Y
- 과세표준: 3억원
- 기본세율: 35%

**처리**:
- 중과세율 = 35% + 30% = 65%
- 양도소득세 = 3억 × 65% - 조정된 누진공제

### 4.5 15년 이상 장기보유
**시나리오**: 30년 보유 후 양도
**예시**:
- 보유기간: 30년
- 주택수: 1

**처리**:
- 장기보유특별공제율 = 80% (최대치)
- 공제율은 15년 이상이어도 80%로 고정

### 4.6 0원 양도차익
**시나리오**: 취득가액 = 양도가액
**예시**:
- 취득가액: 5억원
- 양도가액: 5억원
- 양도차익: 0원

**처리**:
- 양도소득세 = 0원
- 계산 중단 (더 이상 계산 불필요)

### 4.7 필요경비가 양도차익을 초과
**시나리오**: 과다한 필요경비
**예시**:
- 양도가액: 5억원
- 취득가액: 4억원
- 필요경비: 1.5억원
- 양도차익: 5억 - 4억 - 1.5억 = -0.5억 (손실)

**처리**:
- 양도손실로 처리
- 세금 0원

### 4.8 동일 연도 복수 양도
**시나리오**: 한 해에 2건 이상 양도
**예시**:
- 1차 양도: 3월 (기본공제 250만원 사용)
- 2차 양도: 9월

**처리**:
- 기본공제는 연 1회만 적용
- 2차 양도 시 기본공제 불가

### 4.9 매우 큰 금액 (1조원 이상)
**시나리오**: 초고가 부동산
**예시**:
- 양도가액: 2조원
- 과세표준: 1.5조원

**처리**:
- JavaScript Number 타입 한계 확인 필요
- BigInt 사용 또는 금액 단위 조정 (억원 단위)
- 최대 세율 45% 적용 (또는 중과 시 75%)

### 4.10 거주기간 포함 계산 (1세대1주택)
**시나리오**: 보유 5년, 거주 3년
**예시**:
- 보유기간: 5년
- 거주기간: 3년 (보유기간 내)
- 총 기간: 5년 + 3년 = 8년

**처리**:
- 장기보유특별공제 계산 시 8년 기준
- 공제율 32% 적용

## 5. 입력 파라미터 정의

### 5.1 필수 입력
```typescript
interface CapitalGainsTaxInput {
  // 기본 정보
  acquisitionDate: string;        // YYYY-MM-DD
  acquisitionPrice: number;       // 취득가액 (원)
  transferDate: string;           // YYYY-MM-DD
  transferPrice: number;          // 양도가액 (원)

  // 주택 정보
  houseCount: 1 | 2 | 3;         // 1: 1주택, 2: 2주택, 3: 3주택 이상
  isRegulatedArea: boolean;       // 조정대상지역 여부

  // 선택 입력
  necessaryExpenses?: number;     // 필요경비 (기본값: 0)
  residenceYears?: number;        // 거주기간 (년, 1주택인 경우)
}
```

### 5.2 출력 구조
```typescript
interface CapitalGainsTaxResult {
  // 계산 과정
  holdingPeriod: {
    years: number;                // 보유년수
    days: number;                 // 보유일수
  };

  capitalGain: number;            // 양도차익

  exemption: {
    applied: boolean;             // 비과세 적용 여부
    amount: number;               // 비과세 금액
    reason: string;               // 비과세 사유
  };

  longTermDeduction: {
    rate: number;                 // 공제율 (0~0.80)
    amount: number;               // 공제액
    totalYears: number;           // 총 보유년수
  };

  basicDeduction: number;         // 기본공제 (250만원)

  taxBase: number;                // 과세표준

  taxation: {
    basicRate: number;            // 기본세율
    basicTax: number;             // 기본세액
    surchargeApplied: boolean;    // 중과세 적용 여부
    surchargeRate: number;        // 중과세율
    totalRate: number;            // 최종 적용세율
    capitalGainsTax: number;      // 양도소득세
  };

  localIncomeTax: number;         // 지방소득세 (10%)
  totalTax: number;               // 총 납부세액

  // 메타 정보
  calculationDate: string;        // 계산 수행 일자
  warnings: string[];             // 주의사항
}
```
